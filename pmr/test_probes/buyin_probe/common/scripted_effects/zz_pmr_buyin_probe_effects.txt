# PMR foreign buy-in probe (test builds only). common/scripted_effects.
# Logs the Valencia paper mill: level, levels owned by SPA and GBR (government + investors), government share.
pmr_bp_log = {
	s:STATE_VALENCIA.region_state:SPA = {
		set_variable = { name = pmr_bp_lvl value = -1 }
		set_variable = { name = pmr_bp_spa value = -1 }
		set_variable = { name = pmr_bp_gbr value = -1 }
		set_variable = { name = pmr_bp_gov value = -1 }
		if = {
			limit = { has_building = building_paper_mill }
			set_variable = { name = pmr_bp_lvl value = b:building_paper_mill.level }
			set_variable = { name = pmr_bp_spa value = 0 }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 1 } } } set_variable = { name = pmr_bp_spa value = 1 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 2 } } } set_variable = { name = pmr_bp_spa value = 2 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 3 } } } set_variable = { name = pmr_bp_spa value = 3 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 4 } } } set_variable = { name = pmr_bp_spa value = 4 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 5 } } } set_variable = { name = pmr_bp_spa value = 5 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 6 } } } set_variable = { name = pmr_bp_spa value = 6 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 7 } } } set_variable = { name = pmr_bp_spa value = 7 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:SPA value >= 8 } } } set_variable = { name = pmr_bp_spa value = 8 } }
			set_variable = { name = pmr_bp_gbr value = 0 }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 1 } } } set_variable = { name = pmr_bp_gbr value = 1 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 2 } } } set_variable = { name = pmr_bp_gbr value = 2 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 3 } } } set_variable = { name = pmr_bp_gbr value = 3 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 4 } } } set_variable = { name = pmr_bp_gbr value = 4 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 5 } } } set_variable = { name = pmr_bp_gbr value = 5 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 6 } } } set_variable = { name = pmr_bp_gbr value = 6 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 7 } } } set_variable = { name = pmr_bp_gbr value = 7 } }
			if = { limit = { b:building_paper_mill = { levels_owned_by_country = { target = c:GBR value >= 8 } } } set_variable = { name = pmr_bp_gbr value = 8 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.00 } } set_variable = { name = pmr_bp_gov value = 0 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.10 } } set_variable = { name = pmr_bp_gov value = 10 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.20 } } set_variable = { name = pmr_bp_gov value = 20 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.30 } } set_variable = { name = pmr_bp_gov value = 30 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.40 } } set_variable = { name = pmr_bp_gov value = 40 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.50 } } set_variable = { name = pmr_bp_gov value = 50 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.60 } } set_variable = { name = pmr_bp_gov value = 60 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.70 } } set_variable = { name = pmr_bp_gov value = 70 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.80 } } set_variable = { name = pmr_bp_gov value = 80 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 0.90 } } set_variable = { name = pmr_bp_gov value = 90 } }
			if = { limit = { b:building_paper_mill = { country_ownership_fraction >= 1.00 } } set_variable = { name = pmr_bp_gov value = 100 } }
		}
		debug_log = "PMR;rec=BUYIN_PROBE;arm=$ARM$;date=[TimeKeeper.GetCurrentDate.GetString];level=[THIS.Var('pmr_bp_lvl').GetValue|0];spa=[THIS.Var('pmr_bp_spa').GetValue|0];gbr=[THIS.Var('pmr_bp_gbr').GetValue|0];gov_pct=[THIS.Var('pmr_bp_gov').GetValue|0]"
	}
}
