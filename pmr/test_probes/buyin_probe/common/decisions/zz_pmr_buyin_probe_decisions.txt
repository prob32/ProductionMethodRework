# PMR foreign buy-in probe (test builds only). common/decisions. Run IN ORDER in a new 1836 game, one per day.
# Each logs PMR;rec=BUYIN_PROBE;arm=<step>_BEFORE and _AFTER (level, spa, gbr, gov_pct). Also check error.log.
# Step 1. Constants: SPA government 1, GBR district (Home Counties) 1. Expect 3 levels: SPA 2 (gov 1 + district 1), GBR 1.
pmr_buyin_probe_1_decision = {
	is_shown = { always = yes }
	possible = { always = yes }
	when_taken = {
		pmr_bp_log = { ARM = S1_BEFORE }
		s:STATE_VALENCIA.region_state:SPA = {
			create_building = {
				building = building_paper_mill
				add_ownership = {
					country = { country = c:SPA levels = 1 }
					building = { type = building_financial_district country = c:GBR region = STATE_HOME_COUNTIES levels = 1 }
				}
			}
		}
		pmr_bp_log = { ARM = S1_AFTER }
	}
	ai_chance = { value = 0 }
}
