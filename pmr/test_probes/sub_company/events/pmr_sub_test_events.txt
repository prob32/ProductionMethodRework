namespace = pmr_sub_test

# Country scope, hidden: the sub-company smoke test's log every 30 days while a test company exists; the line that
# finds both gone is the last.
pmr_sub_test.1 = {
	type = country_event
	hidden = yes

	immediate = {
		pmr_sub_test_log = yes
		if = {
			limit = {
				OR = {
					has_company = company_type:company_pmr_sub_test_a
					has_company = company_type:company_pmr_sub_test_b
				}
			}
			trigger_event = { id = pmr_sub_test.1 days = 30 }
		}
	}
}
